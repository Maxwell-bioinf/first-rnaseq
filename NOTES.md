## Папки и файлы
mkdir                      # создает папку (Например mkdir -p ~/projects/first-rnaseq создал папку first-rnaseq по пути: ~/projects)
cd                         # Заходит в папку (Например cd ~/projects/fist-rnaseq заходит в созданную выше папку)
printf                     # создать файл в папке (Пример: printf "gene1\t10\ngene2\t250\ngene3\t0\ngene4\t75\n" > test.tsv, не забывай про n!, при этом > заменяет все содержимое файла, а >> дописывает в конец файла)
pwd	                   # где я сейчас
ls	                   # что лежит в папке
cat                        # показать папку целиком
## Просмотр данных в файле
head -n 2	           # вывести первые 2 строки, по умолчанию 10
wc -l	                   # посчитать строки
grep слово файл	           # найти строки со словом
sort -k2 -n -r	           # сортировка по колонке 2 по убыванию
wc --help                  # прочитать помощь (справку)
head --help | less         # справка, выход через q
## Git-комманды
git add                    # добавить файлы\файл в корзину для созания снимка 
git commit -m ""           # закоммитить (сделать снимок корзины) с добавлением комментария типа: что сделано. 
git push                   # отправить закомиченные файлы в гитхаб.
